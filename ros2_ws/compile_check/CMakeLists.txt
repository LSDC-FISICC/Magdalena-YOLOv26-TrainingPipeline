cmake_minimum_required(VERSION 3.14)
project(compile_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../src)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED)
# cameraNode: full build and link (no CUDA/TensorRT needed)
add_executable(cameraNode ${SRC}/cameraNode.cpp)
target_include_directories(cameraNode PRIVATE ${SRC} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(cameraNode rclcpp::rclcpp ${sensor_msgs_TARGETS} cv_bridge::cv_bridge ${OpenCV_LIBS})
# inference node: compile only, TensorRT/CUDA replaced by stub headers
add_library(inference_objs OBJECT ${SRC}/main.cpp ${SRC}/InferenceNode_v2.cpp ${SRC}/model.cpp)
target_include_directories(inference_objs PRIVATE ${CMAKE_SOURCE_DIR}/stubs ${SRC} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(inference_objs rclcpp::rclcpp ${sensor_msgs_TARGETS} ${diagnostic_msgs_TARGETS} cv_bridge::cv_bridge ${OpenCV_LIBS})
target_compile_options(inference_objs PRIVATE -Wall -Wextra)
