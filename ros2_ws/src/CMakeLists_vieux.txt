cmake_minimum_required(VERSION 3.14)
project(run_model_cpp_node LANGUAGES CXX CUDA)

# Configuration standard C++
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# ---------------------------------------------------------------------------
# Recherche des paquets dépendances (ROS 2)
# ---------------------------------------------------------------------------
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
if(TARGET cv_bridge::cv_bridge)
    message(STATUS ">>> cv_bridge::cv_bridge target FOUND")
    get_target_property(CVB_INC cv_bridge::cv_bridge INTERFACE_INCLUDE_DIRECTORIES)
    message(STATUS ">>> cv_bridge include dirs: ${CVB_INC}")
else()
    message(WARNING ">>> cv_bridge::cv_bridge target NOT FOUND")
endif()

message(STATUS ">>> cv_bridge_INCLUDE_DIRS variable: ${cv_bridge_INCLUDE_DIRS}")
message(STATUS ">>> cv_bridge_DIR: ${cv_bridge_DIR}")

# ---------------------------------------------------------------------------
# Recherche des dépendances Système / Vision / Accélération
# ---------------------------------------------------------------------------
find_package(OpenCV REQUIRED)
find_package(CUDAToolkit REQUIRED)

# Recherche de TensorRT
find_path(TENSORRT_INCLUDE_DIR NvInfer.h
    HINTS /usr/include /usr/local/cuda/include /home/maximilien-godin/TensorRT-10.9.0.34/include
)

find_library(TENSORRT_LIBRARY nvinfer
    HINTS /usr/lib /usr/local/cuda/lib64 /home/maximilien-godin/TensorRT-10.9.0.34/lib
)

# ---------------------------------------------------------------------------
# Configuration de l'exécutable unique
# ---------------------------------------------------------------------------
add_executable(inference_node
    main.cpp
    InferenceNode.cpp
    model.cpp
)

# Inclusion explicite des répertoires pour s'assurer que le préprocesseur trouve TOUS les .h/.hpp
target_include_directories(inference_node PRIVATE
    ${PROJECT_SOURCE_DIR}
    ${TENSORRT_INCLUDE_DIR}
    ${cv_bridge_INCLUDE_DIRS}
    ${rclcpp_INCLUDE_DIRS}
    ${sensor_msgs_INCLUDE_DIRS}
    ${OpenCV_INCLUDE_DIRS}
)

# Liaison moderne par cibles (évite les conflits de signature et force l'ordre de compilation)
target_link_libraries(inference_node PRIVATE
    rclcpp::rclcpp
    sensor_msgs::sensor_msgs
    cv_bridge::cv_bridge
    ${OpenCV_LIBS}
    CUDA::cudart
    ${TENSORRT_LIBRARY}
)

target_compile_options(inference_node PRIVATE -Wall -Wextra)

install(TARGETS inference_node
    DESTINATION lib/${PROJECT_NAME}
)

ament_package()
